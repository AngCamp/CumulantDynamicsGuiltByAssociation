# 2. global reduction step
obj.global_pca(n_components=10, whiten=False, standardize=True)
obj.pca_report(show=True)
# Human judgement: decide whether PCA variance is stable, whether to keep more or fewer PCs, whether to inspect another reduction.

# 3. fold construction step
obj.build_folds(n_folds=5, strategy="temporal_block_randomized", seed=0)
# Human judgement: decide whether block-shuffled folds are appropriate for the time series.

# 4. HMM fitting step
obj.fit_hmm(
    k_values=list(range(10, 1, -1)),
    n_iter=100,
    covariance_type="diag",
    verbose=False,
    use_cv=True,
    report="selected",
)
# Human judgement: inspect ranked CV scores and choose a working K.

# 5. HMM reporting step
obj.hmm_report(report="selected")
